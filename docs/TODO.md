```bash
# 零参数：自动选唯一在线设备，正式速度录一次完整往返

python scripts/capture_anim.py --launcher

# 多台设备 / 手动指定入口坐标 / 只录进入
python scripts/capture_anim.py -s GAQOHMNFQWFUPBQO --name cr11_recheck
python scripts/capture_anim.py --tap 1077,263 --no-back
```
